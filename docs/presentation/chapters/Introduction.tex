% More formal definition of clustering
\begin{frame}{What is clustering - More Formally}
    \centering
    \begin{itemize}
        \item Partition a dataset $\mathcal{X}=\{\boldsymbol{x}_1,\boldsymbol{x}_2,\dots, \boldsymbol{x}_N\}$ where $\boldsymbol{x}_i \in \mathbb{R}^D$
        into $K$ groups (clusters).
        \item Each cluster is modelled as a probability distribution with parameters $\theta_k$.
        \item Find the cluster parameters that maximize the log-likelihood of the data:
        \[\max_{\theta} \mathcal{L}(\theta|\mathcal{X})\]
        \[
        \mathcal{L}(\theta|\mathcal{X}) = \sum_{i=1}^{N} \log \left( \sum_{k=1}^{K} \pi_k \,
        p(\boldsymbol{x}_i \mid \theta_k) \right)
        \]
    \end{itemize}
\end{frame}

% Definition of Gaussian Mixture Models
\begin{frame}{Gaussian Mixture Models}
    \centering
    \begin{itemize}
        \item We model each cluster with a Gaussian distribution with its own mean and covariance:        
        \[
                p(\boldsymbol{x} \mid \theta) = \sum_{k=1}^{K} \pi_k \, \mathcal{N}(\boldsymbol{x} \mid \boldsymbol{\mu}_k, \Sigma_k)
        \]
        \[
            \sum_{k=1}^{K} \pi_k = 1, \quad \pi_k \geq 0
        \]
        % PI is the mixture weight: probability that a random point comes from the component k
        \item The probability density function, the likelihood of a data point $\boldsymbol{x}$ being generated by the $k$-th Gaussian component, is given by:
        \[
        \mathcal{N}(\boldsymbol{x} \mid \boldsymbol{\mu}_k, \Sigma_k)
        = \frac{1}{(2\pi)^{D/2} |\Sigma_k|^{1/2}}
        \exp\left( -\frac{1}{2} (\boldsymbol{x} - \boldsymbol{\mu}_k)^T 
        \Sigma_k^{-1} (\boldsymbol{x} - \boldsymbol{\mu}_k) \right)
        \]
    \end{itemize}
\end{frame}

% Approximation of GMM with diagonal covariance matrices
\begin{frame}{Gaussian Mixture Models -- Approximation}
    \centering
    \begin{itemize}
    \item We assumed the features are uncorrelated within each component, simplifying the covariance matrix to a diagonal form:
    \[
    \Sigma_k = 
    \begin{pmatrix}
    \sigma_{k,1} & 0 & \cdots & 0 \\
    0 & \sigma_{k,2} & \cdots & 0 \\
    \vdots & \vdots & \ddots & \vdots \\
    0 & 0 & \cdots & \sigma_{k,D}
    \end{pmatrix}
    \]
    \item This implies that the likelihood can be expressed as:
    \[
    \mathcal{N}(\boldsymbol{x} \mid \boldsymbol{\mu}_k, \Sigma_k)
    = \frac{1}{(2\pi)^{D/2} \prod_{d=1}^D \sigma_{k,d}^{1/2}}
    \exp\Bigg(-\frac{1}{2} \sum_{d=1}^D \frac{(x_d - \mu_{k,d})^2}{\sigma_{k,d}}\Bigg)
    \]
    \end{itemize}
\end{frame}

% Problem statement
\begin{frame}{Chicken and Egg Problem}
    \centering
    \begin{itemize}
        \item To cluster the data, we need to know the parameters of the Gaussian components.
        \item To estimate the parameters, we need to know the cluster assignments of the data points.
        \item This circular dependency is known as the "chicken and egg" problem in clustering.
    \end{itemize}
\end{frame}

% Presentation of the EM-algorithm
\begin{frame}{The solution - EM-Algorithm}
    \centering
    The Expectation-Maximization (EM) algorithm is an iterative method to find maximum likelihood estimates of parameters in models with latent variables.
\end{frame}

% EM-algorithm steps
\begin{frame}{EM-Algorithm - Steps}
    \centering
    The EM algorithm consists of two main steps:
    \vfill
    \begin{itemize}
        \item \textbf{E-step (Expectation step):} Compute the responsibilities, given the current parameter estimates (soft assignments).
        \item \textbf{M-step (Maximization step):} Update the parameters to maximize the expected log-likelihood, using the responsibilities
        computed in the E-step.
    \end{itemize}
\end{frame}

% Pseudocode of the EM-algorithm
\begin{frame}{EM-Algorithm - Pseudocode}
    \begin{algorithm}[H]
    \caption{Expectation-Maximization for GMM clustering}
    \begin{algorithmic}[1]
    \State Randomly initialize parameters $\{\pi_k, \mu_k, \Sigma_k\}_{k=1}^K$
    \For{i $= 1 \dots \text{MAX\_ITER}$}
        \State \textbf{E-step:} compute responsibilities 
        \State \textbf{M-step:} update parameters 
        \State Compute log-likelihood $\mathcal{L}$ 
        \If{Converged} 
            \State break 
        \EndIf
    \EndFor
    \State \textbf{Clustering:} assign labels 
    \end{algorithmic}
    \label{em_alg}
    \end{algorithm}
\end{frame}

% E-step details
\begin{frame}{EM-Algorithm - E-Step}
    \centering
    Using the current parameter estimates $\theta$, we compute the responsibilities, i.e. the posterior probability that data point $i$ belongs to Gaussian component $k$.
    \[
    \gamma_{ik} 
    = \frac{\pi_k \, \mathcal{N}(\boldsymbol{x}_i \mid \boldsymbol{\mu}_k, \Sigma_k)}
        {\sum_{j=1}^{K} \pi_j \, \mathcal{N}(\boldsymbol{x}_i \mid \boldsymbol{\mu}_j, \Sigma_j)}
    \]
    % This gives us a soft assignment of each data point to each cluster.
    % The denominator is used to normalize the responsibilities, allowing the sum over all K to be equal to 1.
\end{frame}

% M-step details
\begin{frame}{EM-Algorithm - M-Step}
    \centering
    Using the responsibilities \(\gamma_{ik}\) computed in the E-step, we update the parameters:
    \vfill
    \begin{itemize}
        \item \textbf{Sum of responsibilities for each component:} \(N_k = \sum_{i=1}^{N} \gamma_{ik}\)
        \item \textbf{Update of means}: \(\boldsymbol{\mu}_k = \frac{1}{N_k} \sum_{i=1}^{N} \gamma_{ik} \boldsymbol{x}_i\)   
        \item \textbf{Update of covariances}: \(\Sigma_k = \frac{1}{N_k} \sum_{i=1}^{N} \gamma_{ik} (\boldsymbol{x}_i - \boldsymbol{\mu}_k)(\boldsymbol{x}_i - \boldsymbol{\mu}_k)^T\)
        \item \textbf{Update of mixture weights}: \(\pi_k = \frac{N_k}{N}\)
    \end{itemize}
\end{frame}

% Formalization of computation of log-likelihood and convergence criteria
\begin{frame}{Convergence}
    \centering
    The log-likelihood of the data given the model parameters is computed as:
    \[
    \mathcal{L}(\theta ) = \sum_{i=1}^{N} \log \left( \sum_{k=1}^{K} \pi_k \, \mathcal{N}(\boldsymbol{x}_i \mid \boldsymbol{\mu}_k, \Sigma_k) \right)
    \]
    Convergence it's reached if the change is below a predefined threshold $\epsilon$.
\end{frame}

% Formalization of clustering assignment
\begin{frame}{Clustering Assignment}
    \centering
    After convergence, each data point is assigned to the cluster with the highest responsibility:
    \[
    \hat{z}_i = \arg \max_{k} p(z_i = k \mid \boldsymbol{x}_i, \theta) = \arg\max_{k} \gamma_{ik}
    \]
\end{frame}

% Computational complexity analysis
\begin{frame}{Computational Complexity Analysis}
    \centering
    \begin{itemize}
        \item The computational cost is expressed in terms of number of data points $N$, clusters $K$, and feature dimensions $D$.
        \item Both E-step and M-step involve operations over all data points and clusters:
        \[O(NKD)\]
        \item Overall iteration cost:
        \[O(NKD) + O(NKD) = O(NKD)\]
    \end{itemize}
\end{frame}
